 const undef=P.components.filter(n=>!def.has(n));if(undef.length)fail('G3 not defined in kept sections: '+undef.join(', '));

const propsOf=async n=>{const src=P.componentMap[n];if(!src)return null;const t=await rd(src.replace(/\.(jsx|tsx|js|ts)$/,'.d.ts'));if(!t)return null;

 const tn=sig?sig[1]:n+'Props';if(/^React\./.test(tn))return {req:[],children:true};
